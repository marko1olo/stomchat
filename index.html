# gemini_client.py — Failover Cascade across Gemini 2.5 and Groq Llama-3 from typing import List, Optional import google.generativeai as genai async def generate_clinical_draft(prompt: str, history: List[dict]) -> str: cascade_models = ["gemini-2.5-flash", "groq/llama-3.3-70b", "gemini-1.5-pro"] for model_name in cascade_models: try: client = get_provider_client(model_name) response = await client.generate_async(prompt, history=history, timeout=4.5) return response.text except Exception as err: logger.warning(f"Provider {model_name} failed ({err}), falling back...") continue raise RuntimeError("All AI cascade providers unavailable")
# visionproc.py — OpenCV & Pillow Radiograph Preprocessing import cv2 from PIL import Image def normalize_dental_radiograph(image_path: str) -> str: """Enhances periapical radiograph contrast using CLAHE.""" img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE) clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8)) enhanced = clahe.apply(img) output_path = image_path.replace(".jpg", "_clahe.jpg") cv2.imwrite(output_path, enhanced) return output_path
